# Format : Code_Binaire -> (En-Phase I, Quadrature Q)
--------------------------------------------------
0000 -> (-3, -3)
0001 -> (-3, -1)
0010 -> (-3, 3)
0011 -> (-3, 1)
0100 -> (-1, -3)
0101 -> (-1, -1)
0110 -> (-1, 3)
0111 -> (-1, 1)
1000 -> (3, -3)
1001 -> (3, -1)
1010 -> (3, 3)
1011 -> (3, 1)
1100 -> (1, -3)
1101 -> (1, -1)
1110 -> (1, 3)
1111 -> (1, 1)
